	\section{研究方法}
	\subsection{物理模型建立：二维伊辛模型}
	本研究采用二维伊辛模型 (2D Ising Model) 来模拟铁磁材料的微观磁性行为。系统由 $N \times N$ 个排列在正方晶格上的自旋 (Spin) 组成。每个格点 $i$ 上的自旋变量 $s_i$ 仅有两个取值：$+1$（自旋向上）或 $-1$（自旋向下）。
	
	系统的总能量由哈密顿量 (Hamiltonian) 描述：
	
	$$H = -J \sum_{\langle i, j \rangle} s_i s_j$$
	
	其中：
	
	$J$ 为交换相互作用常数 (Coupling Constant)。本研究设 $J=1 > 0$，代表铁磁相互作用，即相邻自旋倾向于平行排列（同向能量更低）。$\langle i, j \rangle$ 表示对所有最近邻 (Nearest Neighbors) 进行求和。在二维正方晶格中，每个格点有 4 个最近邻（上、下、左、右）。
	
	\subsection{数值模拟算法：Metropolis-Hastings}
	由于系统的可能状态数随粒子数指数增长 ($2^{N^2}$)，直接计算配分函数在计算上不可行。本研究采用 Metropolis-Hastings 蒙特卡洛算法 (MCMC) 进行重要性采样，以仿真系统在特定温度 $T$ 下的热力学平衡态。
	
	算法具体迭代步骤如下：
	
	初始化：随机生成一个初始自旋组态（高温无序态）。
	
	单点翻转尝试：在晶格中随机选取一个格点 $i$，假设将其自旋翻转 ($s_i \to -s_i$)。
	
	能量差计算：计算翻转前后系统的能量变化 $\Delta E$：
	
	$$\Delta E = E_{new} - E_{old} = 2 J s_i \sum_{neighbors} s_j$$
	
	接受判据 (Metropolis Criterion)：
	
	若 $\Delta E \le 0$：翻转后系统能量降低（趋于稳定），必定接受该翻转。
	
	若 $\Delta E > 0$：翻转后系统能量升高，则以玻尔兹曼概率 $P$ 随机接受该翻转：
	
	$$P = \exp\left(-\frac{\Delta E}{k_B T}\right)$$(本模拟中简化取玻尔兹曼常数 $k_B=1$)。
	
	循环迭代：重复上述步骤，驱动系统向热力学平衡态演化。
	
	\subsection{MATLAB 实现与可视化设计}
	本研究利用 MATLAB 强大的矩阵运算与图形处理功能实现仿真：
	
	交互式参数输入： 利用 inputdlg 函数构建图形用户界面 (GUI)，允许用户在运行时自定义网格尺寸 $N$、温度 $T$ 及总步数，增强了实验的灵活性。
	
	微观场可视化： 使用 quiver 函数绘制二维向量场，以箭头方向直观展示每个格点的自旋状态 ($s_i = \pm 1$)，实现微观结构的动态监测。实时数据监测： 
	
	通过 drawnow 函数实时刷新图像，动态绘制系统平均磁化强度 $M$ 随时间演化的曲线，用于判断相变发生的时刻。

	\section{结论和未来展望}
	\subsection{研究结论}
	本文基于 MATLAB 平台，利用 Metropolis-Hastings 蒙特卡洛算法对二维伊辛模型 (2D Ising Model) 进行了数值模拟，并开发了可视化交互界面，从微观与宏观两个维度探究了铁磁系统的热力学相变规律。主要结论如下：
	
	\begin{enumerate}\item 成功复现热力学二阶相变： 仿真结果清晰地展示了系统在临界温度 $T_c \approx 2.27$ 附近的相变行为。实验数据表明，当 $T > T_c$ 时，热扰动占主导，系统处于无序的顺磁态（总磁化强度 $M \approx 0$）；当 $T < T_c$ 时，交换相互作用占主导，系统自发演化为有序的铁磁态（$M \to 1$）。\item 直观验证了自发对称性破缺： 通过引入微观自旋向量场的可视化技术，本研究捕捉到了系统在低温下从高对称性（无序）向低对称性（有序）坍缩的动态过程。这一现象直观地验证了物理学中「自发对称性破缺」的机制，即系统基态随机选择了某一个特定的磁化方向。\item 能量与熵的竞争机制： 实验对比了不同温度下的稳态结构，揭示了热力学系统演化的本质是能量最小化（有序倾向）与熵最大化（无序倾向）之间的博弈。
	\end{enumerate}
	
	\subsection{不足与未来展望}
	尽管本仿真成功展示了伊辛模型的核心物理图像，但受限于计算资源与时间，仍存在一定局限性。未来的研究可在以下几个方向进行拓展：
	
	\begin{enumerate}\item 
		引入外部磁场 ($B \neq 0$)：目前的模型仅考虑了零场情况。未来可以在哈密顿量中加入外部磁场项 $H = -J \sum s_i s_j - B \sum s_i$，以此研究磁滞回线 (Hysteresis Loop) 现象，仿真磁性存储材料的写入与擦除过程。
	\end{enumerate}

		clc; clear; close all;
		
		%% 1. GUI 用户参数输入 (弹窗设计)
		% ----------------------------------------------------------
		prompt = {'输入网格大小 N (建议 30-50):', ...
			'输入温度 T (临界点约 2.27):', ...
			'输入模拟总步数 (建议 >2000):', ...
			'是否录制影片? (1=是, 0=否):'}; % <--- ✨ [新增 1] 录像选项
		dlgtitle = '伊辛模型参数设置';

		% 默认值 (默认开启录像)

		% 如果用户点击取消，则退出
		if isempty(answer)
		disp('用户取消操作。');
		return;
		end
		
		% 将输入的字符串转换为数字
		N = str2double(answer{1});
		T = str2double(answer{2});
		total_steps = str2double(answer{3});
		is_record = str2double(answer{4}); % <--- ✨ [新增 2] 读取录像标记
		J = 1; 
		
		fprintf('开始模拟: N=%d, T=%.2f, Steps=%d, Video=%d\n', N, T, total_steps, is_record);
		
		%% ✨ [新增 3] 视频文件初始化 ✨
		if is_record == 1
		% 自动命名文件名，例如 Ising_T2.27.avi
		video_filename = ['Ising_T', num2str(T), '.avi'];
		v = VideoWriter(video_filename); 
		v.FrameRate = 15; % 设置播放速度 (数值越大越快)
		open(v);
		fprintf('--> 正在录制影片，保存为: %s \n', video_filename);
		end
		
		% ----------------------------------------------------------
		
		%% 2. 系统初始化
		% 随机初始化网格 (+1 或 -1)
		grid_spin = sign(rand(N, N) - 0.5); 
		grid_spin(grid_spin == 0) = 1; 
		
		% 创建一个网格坐标系统，用于画箭头
		[X, Y] = meshgrid(1:N, 1:N);
		
		% 准备绘图窗口
		figure('Name', '高级伊辛模型物理仿真', 'Color', 'white', 'Position', [50, 100, 1400, 500]);
		magnetization_history = zeros(1, total_steps);
		
		%% 3. 主循环 (Metropolis Monte Carlo)
		for step = 1:total_steps
		
		% --- Metropolis 算法核心 ---
		for k = 1 : (N * N) 
		r = randi(N); c = randi(N);
		r_up = mod(r - 2, N) + 1; r_down = mod(r, N) + 1;
		c_left = mod(c - 2, N) + 1; c_right= mod(c, N) + 1;
		
		sum_neighbors = grid_spin(r_up, c) + grid_spin(r_down, c) + ...
		grid_spin(r, c_left) + grid_spin(r, c_right);
		
		dE = 2 * J * grid_spin(r, c) * sum_neighbors;
		
		if dE <= 0 || rand() < exp(-dE / T)
		grid_spin(r, c) = -grid_spin(r, c);
		end
		end
		
		% --- 记录数据 ---
		magnetization_history(step) = abs(sum(grid_spin(:))) / (N * N);
		
		% --- 高级实时可视化 (每 20 步刷新一次) ---
		if mod(step, 20) == 0 || step == 1
		% [图1：宏观色块图]
		subplot(1, 3, 1);
		imagesc(grid_spin); 
		colormap(jet(2)); 
		title(['宏观磁畴 (Step: ', num2str(step), ')'], 'FontSize', 12);
		axis square; axis off;
		
		% [图2：微观自旋箭头图]
		subplot(1, 3, 2);
		U = zeros(N, N); 
		V = grid_spin;   
		quiver(X, Y, U, V, 0.6, 'k', 'LineWidth', 1); 
		xlim([0.5 N+0.5]); ylim([0.5 N+0.5]);
		title('微观自旋组态 (箭头指向)', 'FontSize', 12);
		axis square; 
		set(gca, 'YDir', 'reverse'); 
		box on; grid off;
		
		% [图3：数据曲线图]
		subplot(1, 3, 3);
		plot(1:step, magnetization_history(1:step), 'r-', 'LineWidth', 1.5);
		xlim([0 total_steps]); ylim([0 1.1]);
		xlabel('时间步', 'FontSize', 10);
		ylabel('总磁化强度 |M|', 'FontSize', 10);
		title(['相变过程 (T = ', num2str(T), ')'], 'FontSize', 12);
		grid on;
		
		drawnow; 
		
		%% ✨ [新增 4] 写入影片帧 ✨
		if is_record == 1
		frame = getframe(gcf); % 抓取当前窗口画面
		writeVideo(v, frame);  % 写入影片
		end
		end
		end
		
		%% ✨ [新增 5] 收尾工作 ✨
		if is_record == 1
		close(v); % 关闭影片文件，保存数据
		msgbox(['模拟完成！影片已保存为: ', video_filename], '成功');
		else
		msgbox('模拟完成！请截图保存结果图。', '成功');
		end
		
		disp('模拟完成！');
