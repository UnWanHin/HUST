#include <map>
std::map<int,int> flagdfs;
void dfs(ALGraph G,int t,void (*visit)(VertexType)){
    visit(G.vertices[t].data);
    flagdfs[t]=1;
    ArcNode *p = G.vertices[t].firstarc;
    while(p!=NULL){
        if(flagdfs[p->adjvex]==0) dfs(G,p->adjvex,visit);
        p=p->nextarc;
    }
}
